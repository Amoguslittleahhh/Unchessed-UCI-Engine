    12	pub fn parse(fen: &str) -> Result<Position, String> {
    13	    let mut pos = Position::empty();
    14	    let mut parts = fen.split_whitespace();
    15	
    16	    let placement = parts.next().ok_or("empty FEN")?;
    17	    let mut rank: i32 = 7;
    18	    let mut file: i32 = 0;
    19	    for ch in placement.chars() {
    20	        match ch {
    21	            '/' => {
    22	                if file != 8 {
    23	                    return Err(format!("rank {} has {} files, not 8", rank, file));
    24	                }
    25	                rank -= 1;
    26	                file = 0;
    27	                if rank < 0 {
    28	                    return Err("too many ranks".into());
    29	                }
    30	            }
    31	            '1'..='8' => {
    32	                file += ch as i32 - '0' as i32;
    33	                if file > 8 {
    34	                    return Err(format!("rank overflow at '{}'", ch));
    35	                }
    36	            }
    37	            _ => {
    38	                if file > 7 {
    39	                    return Err(format!("rank overflow at '{}'", ch));
    40	                }
    41	                let color = if ch.is_ascii_uppercase() {
    42	                    Color::White
    43	                } else {
    44	                    Color::Black
    45	                };
    46	                let piece = match ch.to_ascii_lowercase() {
    47	                    'p' => PAWN,
    48	                    'n' => KNIGHT,
    49	                    'b' => BISHOP,
    50	                    'r' => ROOK,
    51	                    'q' => QUEEN,
    52	                    'k' => KING,
    53	                    _ => return Err(format!("bad piece '{}'", ch)),
    54	                };
    55	                let s = sq(file as u8, rank as u8);
    56	                let b = 1u64 << s;
    57	                pos.bb[color.idx()][piece] |= b;
    58	                pos.occ_side[color.idx()] |= b;
    59	                pos.occ |= b;
    60	                pos.board[s as usize] = (color.idx() * 6 + piece) as u8;
    61	                file += 1;
    62	            }
    63	        }
    64	    }
    65	    if file != 8 {
    66	        return Err(format!("rank {} has {} files, not 8", rank, file));
    67	    }
    68	    if rank != 0 {
    69	        return Err(format!("{} ranks given, not 8", 8 - rank));
    70	    }
    71	
    72	    pos.side = match parts.next().unwrap_or("w") {
    73	        "w" => Color::White,
    74	        "b" => Color::Black,
    75	        other => return Err(format!("bad side '{}'", other)),
    76	    };
    77	
    78	    pos.castling = 0;
    79	    if let Some(c) = parts.next() {
    80	        if c != "-" {
    81	            for ch in c.chars() {
    82	                pos.castling |= match ch {
    83	                    'K' => WK,
    84	                    'Q' => WQ,
    85	                    'k' => BK,
    86	                    'q' => BQ,
    87	                    _ => return Err(format!("bad castling '{}'", ch)),
    88	                };
    89	            }
    90	        }
    91	    }
    92	    // A castling right is meaningless (and downstream code assumes it's
    93	    // backed by a real king+rook pair on their home squares -- movegen
    94	    // builds the castling move purely from the rights bit plus path/attack
    95	    // checks, and make_move() unconditionally relocates "the" king and rook
    96	    // there) unless that king and rook actually exist. Reject up front
    97	    // rather than letting a later synthetic castling move panic trying to
    98	    // pick up a piece that was never placed.
    99	    let king_rook_ok = |king_sq: u8, rook_sq: u8, color: Color| {
   100	        pos.piece_on(king_sq) == Some((color, KING)) && pos.piece_on(rook_sq) == Some((color, ROOK))
   101	    };
   102	    if pos.castling & WK != 0 && !king_rook_ok(sq(4, 0), sq(7, 0), Color::White) {
   103	        return Err("castling right K without king/rook on e1/h1".into());
   104	    }
   105	    if pos.castling & WQ != 0 && !king_rook_ok(sq(4, 0), sq(0, 0), Color::White) {
   106	        return Err("castling right Q without king/rook on e1/a1".into());
   107	    }
   108	    if pos.castling & BK != 0 && !king_rook_ok(sq(4, 7), sq(7, 7), Color::Black) {
   109	        return Err("castling right k without king/rook on e8/h8".into());
   110	    }
   111	    if pos.castling & BQ != 0 && !king_rook_ok(sq(4, 7), sq(0, 7), Color::Black) {
   112	        return Err("castling right q without king/rook on e8/a8".into());
   113	    }
   114	
   115	    pos.ep = NO_EP;
   116	    if let Some(e) = parts.next() {
   117	        if e != "-" {
   118	            let ep_sq = parse_sq(e).ok_or_else(|| format!("bad ep '{}'", e))?;
   119	            // The ep target is the square a double-moving pawn passed over,
   120	            // so it must sit on rank 3 (mover was Black) or rank 6 (mover
   121	            // was White), and that mover's pawn must actually be standing
   122	            // one rank behind the target -- otherwise movegen can emit an
   123	            // "en passant" capture with no real pawn behind it, which
   124	            // board.rs's make_move() removes unconditionally, corrupting
   125	            // occupancy/hash instead of erroring.
   126	            let (expected_rank, pawn_rank, mover) = match pos.side {
   127	                Color::White => (5u8, 4u8, Color::Black),
   128	                Color::Black => (2u8, 3u8, Color::White),
   129	            };
   130	            if rank_of(ep_sq) != expected_rank {
   131	                return Err(format!("ep square '{}' not on the rank a double move passes over", e));
   132	            }
   133	            let pawn_sq = sq(file_of(ep_sq), pawn_rank);
   134	            if pos.piece_on(pawn_sq) != Some((mover, PAWN)) {
   135	                return Err(format!("ep square '{}' has no pawn to capture", e));
   136	            }
   137	            if pos.piece_on(ep_sq).is_some() {
   138	                return Err(format!("ep square '{}' is occupied", e));
   139	            }
   140	            pos.ep = ep_sq;
   141	        }
   142	    }
   143	
   144	    pos.halfmove = parts.next().and_then(|s| s.parse().ok()).unwrap_or(0);
   145	    pos.fullmove = parts.next().and_then(|s| s.parse().ok()).unwrap_or(1);
   146	
   147	    if pos.bb[0][KING].count_ones() != 1 || pos.bb[1][KING].count_ones() != 1 {
   148	        return Err("each side needs exactly one king".into());
   149	    }
   150	
   151	    // The side NOT to move must not be in check -- otherwise the side to move
   152	    // could simply capture that king, which is impossible in a real game (the
   153	    // opponent would have had to already be in check on their own turn and
   154	    // failed to address it). This also catches the degenerate "adjacent
   155	    // kings" case, since a king itself attacks all its neighboring squares.
   156	    // Without this check, downstream code that assumes every king square is
   157	    // a normal, uncapturable king (e.g. NNUE king-bucket lookups) can be fed
   158	    // a position where the "opponent" king is effectively already gone,
   159	    // which previously caused an out-of-bounds panic instead of a UCI error.
   160	    let non_mover = pos.side.flip();
   161	    let non_mover_king = pos.bb[non_mover.idx()][KING].trailing_zeros() as u8;
   162	    if attacked(&pos, non_mover_king, pos.side) {
   163	        return Err("illegal position: side not to move is in check".into());
   164	    }
   165	
   166	    pos.hash = pos.compute_hash();
   167	    Ok(pos)
   168	}

   958	fn parse_go(line: &str) -> Limits {
   959	    let mut l = Limits::default();
   960	    let mut toks = line.split_whitespace();
   961	    toks.next();
   962	    while let Some(t) = toks.next() {
   963	        let mut num = |l: &mut Option<u64>| {
   964	            if let Some(v) = toks.next().and_then(|v| v.parse().ok()) {
   965	                *l = Some(v);
   966	            }
   967	        };
   968	        match t {
   969	            "depth" => {
   970	                if let Some(v) = toks.next().and_then(|v| v.parse().ok()) {
   971	                    l.depth = Some(v);
   972	                }
   973	            }
   974	            "movetime" => num(&mut l.movetime),
   975	            "wtime" => num(&mut l.wtime),
   976	            "btime" => num(&mut l.btime),
   977	            "winc" => num(&mut l.winc),
   978	            "binc" => num(&mut l.binc),
   979	            "movestogo" => num(&mut l.movestogo),
   980	            "nodes" => num(&mut l.nodes),
   981	            "infinite" => l.infinite = true,
   982	            _ => {}
   983	        }
   984	    }
   985	    l
   986	}

   988	/// Opponent moves played since we last looked, with their pre-move positions.
   989	fn collect_pending(game: &mut Game) -> Vec<PendingObs> {
   990	    let our_side = game.current.side;
   991	    let n_moves = game.positions.len() - 1;
   992	    let mut out = Vec::new();
   993	    for i in game.observed_plies..n_moves {
   994	        let pre = game.positions[i];
   995	        if pre.side != our_side {
   996	            // reconstruct the move from pre -> post
   997	            let post = game.positions[i + 1];
   998	            if let Some(mv) = legal(&pre)
   999	                .as_slice()
  1000	                .iter()
  1001	                .copied()
  1002	                .find(|m| pre.make(*m).hash == post.hash)
  1003	            {
  1004	                out.push(PendingObs {
  1005	                    pre,
  1006	                    mv,
  1007	                    ply: (i + 1) as u32,
  1008	                });
  1009	            }

  1065	fn prepare_unarchitectured_root_hints(
  1066	    candidate: &Arc<Mutex<Option<UnarchitecturedCandidate>>>,
  1067	    pos: &Position,
  1068	    legal_moves: &[Move],
  1069	    rating: i32,
  1070	    limits: &Limits,
  1071	    minimum_time_ms: u64,
  1072	    exit: InferenceExit,
  1073	) -> PreparedRootHints {
  1074	    if !unarchitectured_wait_allowed(limits, pos.side, minimum_time_ms) {
  1075	        return PreparedRootHints {
  1076	            hints: Vec::new(),
  1077	            elapsed: std::time::Duration::ZERO,
  1078	            source: "skipped-low-time",
  1079	        };
  1080	    }
  1081	    let started = std::time::Instant::now();
  1082	    let input = unarchitectured_input(pos, legal_moves, rating);
  1083	    let key = HintKey::new(pos.hash, &input, exit);
  1084	
  1085	    let exact = {
  1086	        let guard = candidate.lock().unwrap();
  1087	        guard
  1088	            .as_ref()
  1089	            .and_then(|candidate| candidate.worker.latest_exact(&key))
  1090	    };
  1091	    let hint = if exact.is_some() {
  1092	        exact
  1093	    } else {
  1094	        {
  1095	            let guard = candidate.lock().unwrap();
  1096	            if let Some(candidate) = guard.as_ref() {
  1097	                let _ = candidate.worker.try_submit(pos.hash, input, exit);
  1098	            }
  1099	        }
  1100	        let deadline = std::time::Instant::now() + std::time::Duration::from_millis(100);
  1101	        loop {
  1102	            let ready = {
  1103	                let guard = candidate.lock().unwrap();
  1104	                guard
  1105	                    .as_ref()
  1106	                    .and_then(|candidate| candidate.worker.latest_exact(&key))
  1107	            };
  1108	            if ready.is_some() || std::time::Instant::now() >= deadline {
  1109	                break ready;
  1110	            }
  1111	            std::thread::yield_now();
  1112	        }
  1113	    };
  1114	
  1115	    let Some(hint) = hint else {
  1116	        return PreparedRootHints {
  1117	            hints: Vec::new(),
  1118	            elapsed: started.elapsed(),
  1119	            source: "timeout",
  1120	        };
  1121	    };
  1122	    // `zip` would silently truncate to the shorter side, so a logit vector
  1123	    // that did not correspond 1:1 with `legal_moves` would quietly produce a
  1124	    // *partial* hint list -- some legal moves unscored, and (worse) scores
  1125	    // potentially attached to the wrong moves if the orders ever diverged.
  1126	    // Nothing downstream could detect that: the search would just receive a
  1127	    // plausible-looking ranking that is wrong.
  1128	    //
  1129	    // The alignment does hold today, because `HintKey` includes the full
  1130	    // `legal_actions` vector and `latest_exact` only returns a hint whose key
  1131	    // matches exactly. But that is an invariant maintained at a distance, in
  1132	    // a different module, by code that has no obligation to keep doing so.
  1133	    // Assert it here, where the assumption is actually used: on a mismatch,
  1134	    // drop the hint and search unhinted rather than order on bad data.
  1135	    if hint.output.logits.len() != legal_moves.len() {
  1136	        return PreparedRootHints {
  1137	            hints: Vec::new(),
  1138	            elapsed: started.elapsed().max(hint.elapsed),
  1139	            source: "length-mismatch",
  1140	        };
  1141	    }
  1142	    let hints = legal_moves
  1143	        .iter()
  1144	        .zip(hint.output.logits.iter())
  1145	        .map(|(&mv, &policy_score)| search::RootHint { mv, policy_score })
  1146	        .collect();
  1147	    PreparedRootHints {
  1148	        hints,
  1149	        // Charge both request/wait overhead and at least the complete neural
  1150	        // forward to the real search deadline.
  1151	        elapsed: started.elapsed().max(hint.elapsed),
  1152	        source: "exact",
  1153	    }
  1154	}
