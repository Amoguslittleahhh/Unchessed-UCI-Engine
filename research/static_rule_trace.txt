    12	pub fn parse(fen: &str) -> Result<Position, String> {
    13	    let mut pos = Position::empty();
    14	    let mut parts = fen.split_whitespace();
    15	
    16	    let placement = parts.next().ok_or("empty FEN")?;
    17	    let mut rank: i32 = 7;
    18	    let mut file: i32 = 0;
    19	    for ch in placement.chars() {
    20	        match ch {
    21	            '/' => {
    22	                if file != 8 {
    23	                    return Err(format!("rank {} has {} files, not 8", rank, file));
    24	                }
    25	                rank -= 1;
    26	                file = 0;
    27	                if rank < 0 {
    28	                    return Err("too many ranks".into());
    29	                }
    30	            }
    31	            '1'..='8' => {
    32	                file += ch as i32 - '0' as i32;
    33	                if file > 8 {
    34	                    return Err(format!("rank overflow at '{}'", ch));
    35	                }
    36	            }
    37	            _ => {
    38	                if file > 7 {
    39	                    return Err(format!("rank overflow at '{}'", ch));
    40	                }
    41	                let color = if ch.is_ascii_uppercase() {
    42	                    Color::White
    43	                } else {
    44	                    Color::Black
    45	                };
    46	                let piece = match ch.to_ascii_lowercase() {
    47	                    'p' => PAWN,
    48	                    'n' => KNIGHT,
    49	                    'b' => BISHOP,
    50	                    'r' => ROOK,
    51	                    'q' => QUEEN,
    52	                    'k' => KING,
    53	                    _ => return Err(format!("bad piece '{}'", ch)),
    54	                };
    55	                let s = sq(file as u8, rank as u8);
    56	                let b = 1u64 << s;
    57	                pos.bb[color.idx()][piece] |= b;
    58	                pos.occ_side[color.idx()] |= b;
    59	                pos.occ |= b;
    60	                pos.board[s as usize] = (color.idx() * 6 + piece) as u8;
    61	                file += 1;
    62	            }
    63	        }
    64	    }
    65	    if file != 8 {
    66	        return Err(format!("rank {} has {} files, not 8", rank, file));
    67	    }
    68	    if rank != 0 {
    69	        return Err(format!("{} ranks given, not 8", 8 - rank));
    70	    }
    71	
    72	    pos.side = match parts.next().unwrap_or("w") {
    73	        "w" => Color::White,
    74	        "b" => Color::Black,
    75	        other => return Err(format!("bad side '{}'", other)),
    76	    };
    77	
    78	    pos.castling = 0;
    79	    if let Some(c) = parts.next() {
    80	        if c != "-" {
    81	            for ch in c.chars() {
    82	                pos.castling |= match ch {
    83	                    'K' => WK,
    84	                    'Q' => WQ,
    85	                    'k' => BK,
    86	                    'q' => BQ,
    87	                    _ => return Err(format!("bad castling '{}'", ch)),
    88	                };
    89	            }
    90	        }
    91	    }
    92	    // A castling right is meaningless (and downstream code assumes it's
    93	    // backed by a real king+rook pair on their home squares -- movegen
    94	    // builds the castling move purely from the rights bit plus path/attack
    95	    // checks, and make_move() unconditionally relocates "the" king and rook
    96	    // there) unless that king and rook actually exist. Reject up front
    97	    // rather than letting a later synthetic castling move panic trying to
    98	    // pick up a piece that was never placed.
    99	    let king_rook_ok = |king_sq: u8, rook_sq: u8, color: Color| {
   100	        pos.piece_on(king_sq) == Some((color, KING)) && pos.piece_on(rook_sq) == Some((color, ROOK))
   101	    };
   102	    if pos.castling & WK != 0 && !king_rook_ok(sq(4, 0), sq(7, 0), Color::White) {
   103	        return Err("castling right K without king/rook on e1/h1".into());
   104	    }
   105	    if pos.castling & WQ != 0 && !king_rook_ok(sq(4, 0), sq(0, 0), Color::White) {
   106	        return Err("castling right Q without king/rook on e1/a1".into());
   107	    }
   108	    if pos.castling & BK != 0 && !king_rook_ok(sq(4, 7), sq(7, 7), Color::Black) {
   109	        return Err("castling right k without king/rook on e8/h8".into());
   110	    }
   111	    if pos.castling & BQ != 0 && !king_rook_ok(sq(4, 7), sq(0, 7), Color::Black) {
   112	        return Err("castling right q without king/rook on e8/a8".into());
   113	    }
   114	
   115	    pos.ep = NO_EP;
   116	    if let Some(e) = parts.next() {
   117	        if e != "-" {
   118	            let ep_sq = parse_sq(e).ok_or_else(|| format!("bad ep '{}'", e))?;
   119	            // The ep target is the square a double-moving pawn passed over,
   120	            // so it must sit on rank 3 (mover was Black) or rank 6 (mover
   121	            // was White), and that mover's pawn must actually be standing
   122	            // one rank behind the target -- otherwise movegen can emit an
   123	            // "en passant" capture with no real pawn behind it, which
   124	            // board.rs's make_move() removes unconditionally, corrupting
   125	            // occupancy/hash instead of erroring.
   126	            let (expected_rank, pawn_rank, mover) = match pos.side {
   127	                Color::White => (5u8, 4u8, Color::Black),
   128	                Color::Black => (2u8, 3u8, Color::White),
   129	            };
   130	            if rank_of(ep_sq) != expected_rank {
   131	                return Err(format!("ep square '{}' not on the rank a double move passes over", e));
   132	            }
   133	            let pawn_sq = sq(file_of(ep_sq), pawn_rank);
   134	            if pos.piece_on(pawn_sq) != Some((mover, PAWN)) {
   135	                return Err(format!("ep square '{}' has no pawn to capture", e));
   136	            }
   137	            if pos.piece_on(ep_sq).is_some() {
   138	                return Err(format!("ep square '{}' is occupied", e));
   139	            }
   140	            pos.ep = ep_sq;
   141	        }
   142	    }
   143	
   144	    pos.halfmove = parts.next().and_then(|s| s.parse().ok()).unwrap_or(0);
   145	    pos.fullmove = parts.next().and_then(|s| s.parse().ok()).unwrap_or(1);
   146	
   147	    if pos.bb[0][KING].count_ones() != 1 || pos.bb[1][KING].count_ones() != 1 {
   148	        return Err("each side needs exactly one king".into());
   149	    }
   150	
   151	    // The side NOT to move must not be in check -- otherwise the side to move
   152	    // could simply capture that king, which is impossible in a real game (the
   153	    // opponent would have had to already be in check on their own turn and
   154	    // failed to address it). This also catches the degenerate "adjacent
   155	    // kings" case, since a king itself attacks all its neighboring squares.
   156	    // Without this check, downstream code that assumes every king square is
   157	    // a normal, uncapturable king (e.g. NNUE king-bucket lookups) can be fed
   158	    // a position where the "opponent" king is effectively already gone,
   159	    // which previously caused an out-of-bounds panic instead of a UCI error.
   160	    let non_mover = pos.side.flip();
   161	    let non_mover_king = pos.bb[non_mover.idx()][KING].trailing_zeros() as u8;
   162	    if attacked(&pos, non_mover_king, pos.side) {
   163	        return Err("illegal position: side not to move is in check".into());
   164	    }
   165	
   166	    pos.hash = pos.compute_hash();
   167	    Ok(pos)
   168	}
   169	
   170	pub fn serialize(pos: &Position) -> String {
   171	    let chars = ['P', 'N', 'B', 'R', 'Q', 'K'];
   172	    let mut out = String::new();
   173	    for r in (0..8u8).rev() {
   174	        let mut empty = 0;
   175	        for f in 0..8u8 {
   176	            match pos.piece_on(sq(f, r)) {
   177	                Some((c, p)) => {
   178	                    if empty > 0 {
   179	                        out.push_str(&empty.to_string());
   180	                        empty = 0;
   181	                    }
   182	                    let ch = chars[p];
   183	                    out.push(if let Color::White = c {
   184	                        ch
   185	                    } else {
   186	                        ch.to_ascii_lowercase()
   187	                    });
   188	                }
   189	                None => empty += 1,
   190	            }
   191	        }
   192	        if empty > 0 {
   193	            out.push_str(&empty.to_string());
   194	        }
   195	        if r > 0 {
   196	            out.push('/');
   197	        }
   198	    }
   199	    out.push(' ');
   200	    out.push(if let Color::White = pos.side { 'w' } else { 'b' });
   201	    out.push(' ');
   202	    if pos.castling == 0 {
   203	        out.push('-');
   204	    } else {
   205	        if pos.castling & WK != 0 {
   206	            out.push('K');
   207	        }
   208	        if pos.castling & WQ != 0 {
   209	            out.push('Q');
   210	        }

    11	// Direction indices for RAYS. Positive = target index increases along ray.
    12	pub const DIR_N: usize = 0; // +8
    13	pub const DIR_NE: usize = 1; // +9
    14	pub const DIR_E: usize = 2; // +1
    15	pub const DIR_SE: usize = 3; // -7
    16	pub const DIR_S: usize = 4; // -8
    17	pub const DIR_SW: usize = 5; // -9
    18	pub const DIR_W: usize = 6; // -1
    19	pub const DIR_NW: usize = 7; // +7

    21	const DIR_DELTAS: [(i32, i32); 8] = [
    22	    (0, 1),
    23	    (1, 1),
    24	    (1, 0),
    25	    (1, -1),
    26	    (0, -1),
    27	    (-1, -1),
    28	    (-1, 0),
    29	    (-1, 1),
    30	];

    32	const fn build_rays() -> [[Bitboard; 64]; 8] {
    33	    let mut rays = [[0u64; 64]; 8];
    34	    let mut d = 0;
    35	    while d < 8 {
    36	        let (df, dr) = DIR_DELTAS[d];
    37	        let mut s = 0i32;
    38	        while s < 64 {
    39	            let mut f = (s & 7) + df;
    40	            let mut r = (s >> 3) + dr;
    41	            let mut bb = 0u64;
    42	            while f >= 0 && f < 8 && r >= 0 && r < 8 {
    43	                bb |= 1u64 << (r * 8 + f);
    44	                f += df;
    45	                r += dr;
    46	            }
    47	            rays[d][s as usize] = bb;
    48	            s += 1;
    49	        }
    50	        d += 1;
    51	    }
    52	    rays
    53	}

    55	const fn build_leaper(deltas: &[(i32, i32)]) -> [Bitboard; 64] {
    56	    let mut t = [0u64; 64];
    57	    let mut s = 0i32;
    58	    while s < 64 {
    59	        let f0 = s & 7;
    60	        let r0 = s >> 3;
    61	        let mut bb = 0u64;
    62	        let mut i = 0;
    63	        while i < deltas.len() {
    64	            let f = f0 + deltas[i].0;
    65	            let r = r0 + deltas[i].1;
    66	            if f >= 0 && f < 8 && r >= 0 && r < 8 {
    67	                bb |= 1u64 << (r * 8 + f);
    68	            }
    69	            i += 1;
    70	        }
    71	        t[s as usize] = bb;
    72	        s += 1;
    73	    }
    74	    t
    75	}
    76	
    77	const fn build_pawn_attacks() -> [[Bitboard; 64]; 2] {
    78	    let mut t = [[0u64; 64]; 2];
    79	    let mut s = 0i32;
    80	    while s < 64 {
    81	        let f = s & 7;
    82	        let r = s >> 3;
    83	        let mut w = 0u64;
    84	        let mut b = 0u64;
    85	        if r + 1 < 8 {
    86	            if f - 1 >= 0 {
    87	                w |= 1u64 << ((r + 1) * 8 + f - 1);
    88	            }
    89	            if f + 1 < 8 {
    90	                w |= 1u64 << ((r + 1) * 8 + f + 1);
    91	            }
    92	        }
    93	        if r - 1 >= 0 {
    94	            if f - 1 >= 0 {
    95	                b |= 1u64 << ((r - 1) * 8 + f - 1);
    96	            }
    97	            if f + 1 < 8 {
    98	                b |= 1u64 << ((r - 1) * 8 + f + 1);
    99	            }
   100	        }
   101	        t[0][s as usize] = w;
   102	        t[1][s as usize] = b;
   103	        s += 1;
   104	    }
   105	    t
   106	}

   138	// ---------------------------------------------------------------------------
   139	// Slider attacks: classical ray approach (kept as the correctness oracle
   140	// magic-table construction is verified against, and as a fallback reference
   141	// in tests) plus the magic-bitboard tables actually used by search.
   142	// ---------------------------------------------------------------------------
   143	
   144	#[inline]
   145	fn ray_pos(d: usize, s: u8, occ: Bitboard) -> Bitboard {
   146	    let ray = RAYS[d][s as usize];
   147	    let blockers = ray & occ;
   148	    if blockers != 0 {
   149	        let first = blockers.trailing_zeros() as usize;
   150	        ray ^ RAYS[d][first]
   151	    } else {
   152	        ray
   153	    }
   154	}
   155	
   156	#[inline]
   157	fn ray_neg(d: usize, s: u8, occ: Bitboard) -> Bitboard {
   158	    let ray = RAYS[d][s as usize];
   159	    let blockers = ray & occ;
   160	    if blockers != 0 {
   161	        let first = (63 - blockers.leading_zeros()) as usize;
   162	        ray ^ RAYS[d][first]
   163	    } else {
   164	        ray
   165	    }
   166	}
   167	
   168	#[inline]
   169	fn bishop_att_classical(s: u8, occ: Bitboard) -> Bitboard {
   170	    ray_pos(DIR_NE, s, occ) | ray_pos(DIR_NW, s, occ) | ray_neg(DIR_SE, s, occ) | ray_neg(DIR_SW, s, occ)
   171	}
   172	
   173	#[inline]
   174	fn rook_att_classical(s: u8, occ: Bitboard) -> Bitboard {
   175	    ray_pos(DIR_N, s, occ) | ray_pos(DIR_E, s, occ) | ray_neg(DIR_S, s, occ) | ray_neg(DIR_W, s, occ)
   176	}

   178	// ---------------------------------------------------------------------------
   179	// Magic bitboards: O(1) table lookup instead of ray scanning. Magic numbers
   180	// are found at first use (not embedded as literals) by random search,
   181	// verified against the classical functions above as ground truth for every
   182	// occupancy subset -- this makes the search self-checking: it is
   183	// structurally impossible for a "found" magic to produce a wrong attack set,
   184	// since a collision against a differing attack value is rejected and
   185	// retried. Mirrors Stockfish's own approach (it also searches at startup
   186	// rather than embedding a fixed magic table).
   187	// ---------------------------------------------------------------------------
   188	
   189	/// Relevant occupancy mask for a slider on `s`: the full ray in each
   190	/// direction with the outermost (board-edge) square removed, since a piece
   191	/// there can never change the attack set (the ray already terminates at the
   192	/// board edge regardless of whether it's occupied).
   193	fn edge_trim(positive_dir: bool, ray: Bitboard) -> Bitboard {
   194	    if ray == 0 {
   195	        return 0;
   196	    }
   197	    if positive_dir {
   198	        let hi = 63 - ray.leading_zeros();
   199	        ray & !(1u64 << hi)
   200	    } else {
   201	        let lo = ray.trailing_zeros();
   202	        ray & !(1u64 << lo)
   203	    }
   204	}
   205	
   206	fn relevant_bishop_mask(s: u8) -> Bitboard {
   207	    edge_trim(true, RAYS[DIR_NE][s as usize])
   208	        | edge_trim(true, RAYS[DIR_NW][s as usize])
   209	        | edge_trim(false, RAYS[DIR_SE][s as usize])
   210	        | edge_trim(false, RAYS[DIR_SW][s as usize])
   211	}
   212	
   213	fn relevant_rook_mask(s: u8) -> Bitboard {
   214	    edge_trim(true, RAYS[DIR_N][s as usize])
   215	        | edge_trim(true, RAYS[DIR_E][s as usize])
   216	        | edge_trim(false, RAYS[DIR_S][s as usize])
   217	        | edge_trim(false, RAYS[DIR_W][s as usize])
   218	}

   220	#[inline]
   221	fn splitmix64(state: &mut u64) -> u64 {
   222	    *state = state.wrapping_add(0x9E37_79B9_7F4A_7C15);
   223	    let mut z = *state;
   224	    z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
   225	    z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
   226	    z ^ (z >> 31)
   227	}
   228	
   229	/// AND of three random draws biases toward sparse bit patterns, which tend
   230	/// to make better magics (standard trick, same one Stockfish uses).
   231	#[inline]
   232	fn sparse_rand(state: &mut u64) -> u64 {
   233	    splitmix64(state) & splitmix64(state) & splitmix64(state)
   234	}

   236	/// Enumerate every subset of `mask` via the carry-rippler trick (returns
   237	/// 2^popcount(mask) subsets, including 0, terminating back at 0).
   238	fn subsets_of(mask: Bitboard) -> Vec<Bitboard> {
   239	    let mut out = Vec::with_capacity(1 << mask.count_ones());
   240	    let mut subset: Bitboard = 0;
   241	    loop {
   242	        out.push(subset);
   243	        subset = subset.wrapping_sub(mask) & mask;
   244	        if subset == 0 {
   245	            break;
   246	        }
   247	    }
   248	    out
   249	}

   251	#[derive(Clone, Copy)]
   252	struct MagicEntry {
   253	    mask: Bitboard,
   254	    magic: u64,
   255	    shift: u32,
   256	    offset: usize,
   257	}
   258	
   259	struct MagicTables {
   260	    rook: [MagicEntry; 64],
   261	    bishop: [MagicEntry; 64],
   262	    table: Vec<Bitboard>,
   263	}
   264	
   265	/// Search for a working magic for one square + mask, verified against the
   266	/// classical attack function for every occupancy subset of the mask. Two
   267	/// subsets are allowed to collide on the same table index only if they
   268	/// produce the identical attack set; any other collision rejects the magic
   269	/// and retries with a new random candidate.
   270	fn find_magic(
   271	    sq: u8,
   272	    mask: Bitboard,
   273	    classical: fn(u8, Bitboard) -> Bitboard,
   274	    rng: &mut u64,
   275	) -> (u64, Vec<Bitboard>) {
   276	    let bits = mask.count_ones();
   277	    let size = 1usize << bits;
   278	    let shift = 64 - bits;
   279	    let subsets = subsets_of(mask);
   280	    let attacks: Vec<Bitboard> = subsets.iter().map(|&occ| classical(sq, occ)).collect();
   281	
   282	    loop {
   283	        let magic = sparse_rand(rng);
   284	        // quality filter: reject magics that don't spread bits well into the
   285	        // top byte of mask*magic (same heuristic Stockfish uses)
   286	        if (mask.wrapping_mul(magic) >> 56).count_ones() < 6 {
   287	            continue;
   288	        }
   289	        let mut table = vec![0u64; size];
   290	        let mut used = vec![false; size];
   291	        let mut ok = true;
   292	        for (i, &occ) in subsets.iter().enumerate() {
   293	            let idx = ((occ.wrapping_mul(magic)) >> shift) as usize;
   294	            if used[idx] {
   295	                if table[idx] != attacks[i] {
   296	                    ok = false;
   297	                    break;
   298	                }
   299	            } else {
   300	                used[idx] = true;
   301	                table[idx] = attacks[i];
   302	            }
   303	        }
   304	        if ok {
   305	            return (magic, table);
   306	        }
   307	    }
   308	}

   310	fn build_magics() -> MagicTables {
   311	    let mut rng: u64 = 0x00C0_FFEE_D00D_2025;
   312	    let mut table: Vec<Bitboard> = Vec::new();
   313	    let mut rook = [MagicEntry { mask: 0, magic: 0, shift: 0, offset: 0 }; 64];
   314	    let mut bishop = [MagicEntry { mask: 0, magic: 0, shift: 0, offset: 0 }; 64];
   315	
   316	    for s in 0u8..64 {
   317	        let mask = relevant_rook_mask(s);
   318	        let (magic, attacks) = find_magic(s, mask, rook_att_classical, &mut rng);
   319	        let offset = table.len();
   320	        table.extend_from_slice(&attacks);
   321	        rook[s as usize] = MagicEntry { mask, magic, shift: 64 - mask.count_ones(), offset };
   322	    }
   323	    for s in 0u8..64 {
   324	        let mask = relevant_bishop_mask(s);
   325	        let (magic, attacks) = find_magic(s, mask, bishop_att_classical, &mut rng);
   326	        let offset = table.len();
   327	        table.extend_from_slice(&attacks);
   328	        bishop[s as usize] = MagicEntry { mask, magic, shift: 64 - mask.count_ones(), offset };
   329	    }
   330	
   331	    MagicTables { rook, bishop, table }
   332	}

   362	/// Is `s` attacked by side `by`?
   363	pub fn attacked(pos: &Position, s: u8, by: Color) -> bool {
   364	    let b = by.idx();
   365	    // A square is attacked by a `by`-pawn if such a pawn stands on the squares
   366	    // a defender pawn of the other color would attack from `s`.
   367	    if PAWN_ATT[by.flip().idx()][s as usize] & pos.bb[b][PAWN] != 0 {
   368	        return true;
   369	    }
   370	    if KNIGHT_ATT[s as usize] & pos.bb[b][KNIGHT] != 0 {
   371	        return true;
   372	    }
   373	    if KING_ATT[s as usize] & pos.bb[b][KING] != 0 {
   374	        return true;
   375	    }
   376	    let diag = pos.bb[b][BISHOP] | pos.bb[b][QUEEN];
   377	    if diag != 0 && bishop_att(s, pos.occ) & diag != 0 {
   378	        return true;
   379	    }
   380	    let ortho = pos.bb[b][ROOK] | pos.bb[b][QUEEN];
   381	    if ortho != 0 && rook_att(s, pos.occ) & ortho != 0 {
   382	        return true;
   383	    }
   384	    false
   385	}

   148	pub struct Zobrist {
   149	    pub psq: [[[u64; 64]; 6]; 2],
   150	    pub side: u64,
   151	    pub castle: [u64; 16],
   152	    pub ep_file: [u64; 8],

   155	const fn sm_next(state: u64) -> (u64, u64) {
   156	    let s = state.wrapping_add(0x9E37_79B9_7F4A_7C15);
   157	    let mut z = s;
   158	    z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
   159	    z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
   160	    (s, z ^ (z >> 31))
   161	}
   162	
   163	const fn build_zobrist() -> Zobrist {
   164	    let mut st: u64 = 0x00C0_FFEE_D00D_2024;
   165	    let mut psq = [[[0u64; 64]; 6]; 2];
   166	    let mut c = 0;
   167	    while c < 2 {
   168	        let mut p = 0;
   169	        while p < 6 {
   170	            let mut s = 0;
   171	            while s < 64 {
   172	                let (ns, v) = sm_next(st);
   173	                st = ns;
   174	                psq[c][p][s] = v;
   175	                s += 1;
   176	            }
   177	            p += 1;
   178	        }
   179	        c += 1;
   180	    }
   181	    let (ns, side) = sm_next(st);
   182	    st = ns;
   183	    let mut castle = [0u64; 16];
   184	    let mut i = 0;
   185	    while i < 16 {
   186	        let (ns, v) = sm_next(st);
   187	        st = ns;
   188	        castle[i] = v;
   189	        i += 1;
   190	    }
   191	    // castle[0] = 0 so "no rights" contributes nothing; keeps hashes tidy.
   192	    castle[0] = 0;
   193	    let mut ep_file = [0u64; 8];
   194	    let mut i = 0;
   195	    while i < 8 {
   196	        let (ns, v) = sm_next(st);
   197	        st = ns;
   198	        ep_file[i] = v;
   199	        i += 1;
   200	    }
   201	    Zobrist {
   202	        psq,
   203	        side,
   204	        castle,
   205	        ep_file,
   206	    }
   207	}

   209	pub static ZOBRIST: Zobrist = build_zobrist();

   212	const fn build_castle_mask() -> [u8; 64] {
   213	    let mut m = [15u8; 64];
   214	    m[4] = 15 & !(WK | WQ); // e1
   215	    m[0] = 15 & !WQ; // a1
   216	    m[7] = 15 & !WK; // h1
   217	    m[60] = 15 & !(BK | BQ); // e8
   218	    m[56] = 15 & !BQ; // a8
   219	    m[63] = 15 & !BK; // h8
   220	    m
   221	}

   301	        self.occ_side[c.idx()] &= !b;
   302	        self.occ &= !b;

   308	    pub fn compute_hash(&self) -> u64 {
   309	        let mut h = 0u64;
   310	        for s in 0..64u8 {
   311	            if let Some((c, p)) = self.piece_on(s) {
   312	                h ^= ZOBRIST.psq[c.idx()][p][s as usize];
   313	            }
   314	        }
   315	        h ^= ZOBRIST.castle[self.castling as usize];
   316	        if self.ep != NO_EP {
   317	            h ^= ZOBRIST.ep_file[file_of(self.ep) as usize];
   318	        }
   319	        if let Color::Black = self.side {
   320	            h ^= ZOBRIST.side;
   321	        }
   322	        h
   323	    }

   328	        let us = p.side;
   329	        let them = us.flip();
   330	        let from = mv.from();
   331	        let to = mv.to();
   332	        let (_, pt) = p.piece_on(from).expect("make: no piece on from-square");

   335	        if p.ep != NO_EP {
   336	            p.hash ^= ZOBRIST.ep_file[file_of(p.ep) as usize];
   337	            p.ep = NO_EP;
   338	        }
   339	
   340	        let mut capture = false;
   341	        match mv.kind() {
   342	            MK_CASTLE => {
   343	                // king from->to; move the rook too
   344	                let (rf, rt) = if to > from {
   345	                    (to + 1, to - 1)
   346	                } else {
   347	                    (to - 2, to + 1)
   348	                };
   349	                p.take(us, KING, from);
   350	                p.put(us, KING, to);
   351	                p.take(us, ROOK, rf);
   352	                p.put(us, ROOK, rt);
   353	            }
   354	            MK_EP => {
   355	                let cap_sq = if let Color::White = us { to - 8 } else { to + 8 };
   356	                p.take(them, PAWN, cap_sq);
   357	                p.take(us, PAWN, from);
   358	                p.put(us, PAWN, to);
   359	                capture = true;
   360	            }

     6	pub fn perft(pos: &Position, depth: u32) -> u64 {
     7	    if depth == 0 {
     8	        return 1;
     9	    }
    10	    let ml = legal(pos);
    11	    if depth == 1 {
    12	        return ml.len as u64;
    13	    }
    14	    let mut nodes = 0;
    15	    for &m in ml.as_slice() {
    16	        nodes += perft(&pos.make(m), depth - 1);
    17	    }
    18	    nodes
    19	}

    27	            1
    28	        } else {
    29	            perft(&pos.make(m), depth - 1)
    30	        };
    31	        out.push((m.uci(), n));
    32	    }
